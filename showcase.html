# Smart return — closes oldest open checkout
@app.route('/return/<int:book_id>', methods=['POST'])
def return_book(book_id):
    db  = get_db()
    row = db.execute(
        'SELECT id FROM checkouts WHERE book_id = ? AND returned = 0'
        ' ORDER BY checkout_date ASC LIMIT 1',
        (book_id,)
    ).fetchone()
    if row:
        db.execute('UPDATE checkouts SET returned = 1 WHERE id = ?',
                   (row['id'],))
        db.execute('UPDATE books SET available_copies = available_copies + 1'
                   ' WHERE id = ?', (book_id,))
        db.commit()
        flash('Book returned successfully!', 'success')
    return redirect(url_for('index'))
